### 2.1 数据导入模块
```typescript
// 核心流程
```

### 2.2 异常检测规则
```typescript
interface AnomalyRule {
  type: 'camera_lost' | 'data_conflict' | 'format_error';
  check: (record: PocketRecord) => boolean;
  message: string;
}

// 检测规则
const anomalyRules: AnomalyRule[] = [
  {
    type: 'camera_lost',
    check: (r) => !r.cameraAngle || 
               (r.cameraAngle.azimuth === 0 && r.cameraAngle.elevation === 0),
    message: '相机视角参数缺失或为零'
  },
  {
    type: 'data_conflict',
    check: (r) => /* 检测坐标与点云不一致 */ false,
    message: '三维模型与点云切片参数冲突'
  },
  {
    type: 'format_error',
    check: (r) => !r.proteinName || !r.pocketCoordinates,
    message: '必填字段缺失'
  }
];
```

### 2.4 相机视角保存
```typescript
// 使用localStorage持久化视角状态
const CAMERA_STORAGE_KEY = 'pocket_browser_camera_angles';

const saveCameraAngle = (recordId: string, angle: CameraAngle) => {
  const stored = JSON.parse(localStorage.getItem(CAMERA_STORAGE_KEY) || '{}');
  stored[recordId] = angle;
  localStorage.setItem(CAMERA_STORAGE_KEY, JSON.stringify(stored));
};

const restoreCameraAngle = (recordId: string): CameraAngle | null => {
  const stored = JSON.parse(localStorage.getItem(CAMERA_STORAGE_KEY) || '{}');
  return stored[recordId] || null;
};
```

## 3. 状态管理设计

### 3.1 Zustand Store结构
```typescript
interface DataStore {
  // 数据状态
  records: PocketRecord[];
  selectedRecordId: string | null;
  filterConditions: FilterConditions;
  sortConfig: SortConfig;

  // 导入状态
  importProgress: number;
  importErrors: ImportError[];

  // 筛选状态
  statusFilter: ('usable' | 'pending' | 'unusable')[];
  anomalyTypeFilter: string[];
  searchKeyword: string;

  // 操作方法
  importData: (file: File) => Promise<void>;
  updateRecord: (id: string, updates: Partial<PocketRecord>) => void;
  setReviewStatus: (id: string, status: ReviewStatus, note?: string) => void;
  saveCameraAngle: (id: string, angle: CameraAngle) => void;
}
```
